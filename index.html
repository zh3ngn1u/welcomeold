local HttpService = game:GetService("HttpService")
local API_URL = "https://domainanda.com/api/validate"

local function validate(key, hwid)
    local ok, res = pcall(function()
        return HttpService:PostAsync(API_URL,
            HttpService:JSONEncode({ key = key, hwid = hwid }),
            Enum.HttpContentType.ApplicationJson)
    end)
    if not ok then return nil, "koneksi gagal" end
    local data = HttpService:JSONDecode(res)
    if data.success then return data end
    return nil, data.message
end

local data, err = validate("XXXX-XXXX-XXXX-XXXX", "RBX-8f3a12c9e5")
if not data then
    warn("[KeySystem] " .. tostring(err))
    return
end
print("Lisensi valid")
